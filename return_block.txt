  return (

      <div className="min-h-screen flex items-center justify-center text-gray-600">

        <p className="animate-pulse text-lg font-medium">Loading metadata...</p>

      </div>

    );



  return (

    <div className="min-h-screen bg-gray-50 p-6">

      <div className="mx-auto flex w-full max-w-6xl flex-col gap-8">

        <header className="space-y-3 text-center">

          <p className="text-xs uppercase tracking-[0.4em] text-blue-500">Metadata Control Room</p>

          <h1 className="text-3xl font-extrabold bg-gradient-to-r from-blue-700 to-indigo-600 bg-clip-text text-transparent md:text-4xl">

            Salesforce Metadata Explorer

          </h1>

          <p className="text-gray-500">

            Inspect connected org objects, track automation assets, and export field insights with a single view.

          </p>

        </header>



        <section className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">

          {heroStats.map((stat) => (

            <article

              key={stat.label}

              className="rounded-2xl border border-gray-200 bg-white px-4 py-5 text-center shadow-sm"

            >

              <p className="text-xs uppercase tracking-[0.3em] text-gray-400">{stat.label}</p>

              <p className="mt-2 text-3xl font-semibold text-gray-900">{stat.value}</p>

            </article>

          ))}

        </section>



        <section className="space-y-6 rounded-2xl border border-gray-200 bg-white p-6 shadow-sm">

          <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">

            <div>

              <p className="text-xs uppercase tracking-[0.3em] text-gray-400">Objects</p>

              <h2 className="text-2xl font-semibold text-gray-900">Select an object to analyze</h2>

            </div>

            <div className="flex flex-wrap gap-3">

              <button

                type="button"

                className="rounded-full border border-gray-300 px-4 py-2 text-xs font-semibold uppercase tracking-[0.3em] text-gray-600 transition hover:border-blue-500 hover:text-blue-600"

                onClick={() => setObjectSearch('')}

              >

                Clear search

              </button>

              <button

                type="button"

                className="rounded-full bg-blue-600 px-4 py-2 text-xs font-semibold uppercase tracking-[0.3em] text-white shadow-sm shadow-blue-200 transition hover:bg-blue-700"

                onClick={fetchAllData}

              >

                Re-sync data

              </button>

            </div>

          </div>



          <div className="flex flex-col gap-3 md:flex-row md:items-end md:gap-6">

            <label className="flex-1">

              <span className="text-xs font-semibold uppercase tracking-[0.3em] text-gray-500">Search objects</span>

              <input

                type="search"

                placeholder="Type to filter objects"

                value={objectSearch}

                onChange={(e) => setObjectSearch(e.target.value)}

                className="mt-2 w-full rounded-2xl border border-gray-200 bg-white px-4 py-3 text-base text-gray-800 placeholder:text-gray-400 focus:border-blue-500 focus:outline-none focus:ring-2 focus:ring-blue-200"

              />

            </label>

            <div className="flex flex-col gap-1">

              <p className="text-xs uppercase tracking-[0.3em] text-gray-400">Pinned</p>

              <div className="flex flex-wrap gap-2">

                {pinnedObjects.length === 0 ? (

                  <span className="text-xs text-gray-400">Add pins for instant access</span>

                ) : (

                  pinnedObjects.map((obj) => (

                    <span key={obj} className="rounded-full border border-blue-200 bg-blue-50 px-3 py-1 text-xs text-blue-700">

                      {obj}

                    </span>

                  ))

                )}

              </div>

            </div>

          </div>



          <div className="flex flex-wrap gap-2">

            {filteredObjects.length === 0 && selectedObjects.length > 0 ? (

              <p className="text-sm text-gray-500">No objects match that search.</p>

            ) : filteredObjects.length === 0 ? (

              <p className="text-sm text-gray-500">No objects selected in session.</p>

            ) : (

              filteredObjects.map((obj) => (

                <button

                  key={obj}

                  type="button"

                  onClick={() => setSelectedObject(obj)}

                  className={`flex items-center gap-2 rounded-2xl border px-4 py-2 text-sm font-semibold transition ${

                    selectedObject === obj

                      ? 'border-blue-500 bg-blue-50 text-blue-700'

                      : 'border-gray-200 bg-white text-gray-600 hover:border-blue-400'

                    }`}

                >

                  <span>{obj}</span>

                  <span

                    onClick={(e) => {

                      e.stopPropagation();

                      togglePin(obj);

                    }}

                    className={`rounded-full px-2 text-[0.6rem] uppercase tracking-[0.2em] ${

                      pinnedObjects.includes(obj) ? 'bg-blue-100 text-blue-600' : 'bg-gray-100 text-gray-500'

                    }`}

                  >

                    {pinnedObjects.includes(obj) ? 'Pinned' : 'Pin'}

                  </span>

                </button>

              ))

            )}

          </div>



          <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-4">

            {overviewCards.map((card) => (

              <article key={card.label} className="rounded-2xl border border-gray-200 bg-gray-50 px-4 py-5 text-center">

                <p className="text-xs uppercase tracking-[0.3em] text-gray-400">{card.label}</p>

                <p className="mt-2 text-2xl font-semibold text-gray-900">{card.value}</p>

              </article>

            ))}

          </div>

        </section>



        <section className="space-y-6 rounded-2xl border border-gray-200 bg-white p-6 shadow-sm">

          <div className="flex flex-col gap-4 md:flex-row md:items-center md:justify-between">

            <div>

              <p className="text-xs uppercase tracking-[0.3em] text-gray-400">Metadata tabs</p>

              <h2 className="text-2xl font-semibold text-gray-900">Inspect fields and automation</h2>

            </div>

            <div className="flex flex-wrap gap-3">

              {TABS.map((tab) => (

                <button

                  key={tab}

                  type="button"

                  onClick={() => setActiveTab(tab)}

                  className={`rounded-full px-4 py-2 text-xs font-semibold uppercase tracking-[0.3em] transition ${

                    activeTab === tab

                      ? 'bg-gradient-to-r from-blue-600 to-indigo-600 text-white shadow-md'

                      : 'border border-gray-200 bg-white text-gray-600 hover:border-blue-400'

                  }`}

                >

                  {tab}

                </button>

              ))}

            </div>

          </div>

          <div className="space-y-6">

            {activeTab === 'Fields' && (

              <FieldsTable

                selectedObject={selectedObject}

                selectedObjects={selectedObjects}

                fieldsData={fieldsData}

                allMetadataKeys={allMetadataKeys}

              />

            )}

            {activeTab === 'Validation Rules' && (

              <ValidationRulesTable data={validationRules} onRefresh={fetchAllData} />

            )}

            {activeTab === 'Triggers' && (

              <TriggersTable data={triggers} onRefresh={fetchAllData} />

            )}

            {activeTab === 'Apex Classes' && (

              <ApexClassesTable data={apexClasses} onRefresh={fetchAllData} />

            )}

            {activeTab === 'Workflows' && (

              <WorkflowsTable data={workflows} onRefresh={fetchAllData} />

            )}

          </div>

        </section>

      </div>

    </div>

  );





}
